-- ---------------------------------------------------------
-- PART B: CREATE INDEXES - The Optimization
-- These indexes fix the slow queries
-- ---------------------------------------------------------

-- 1. For Q1 (Pending orders) - Foreign key + status filter
CREATE INDEX idx_orders_customer_id ON Orders(customer_id);
CREATE INDEX idx_orders_status ON Orders(order_status);

-- 2. For Q2 (Ankara search) - Most searched column in Shop Didi
CREATE INDEX idx_products_fabric_type ON Products(fabric_type);

-- 3. For Q3 (Custom low stock) - Composite index for 2 conditions
CREATE INDEX idx_products_custom_stock ON Products(is_custom, stock_qty);

-- 4. For Q4 (Sales per designer) - Heavy JOIN needs these
CREATE INDEX idx_products_designer_id ON Products(designer_id);
CREATE INDEX idx_orderitems_product_id ON Order_Items(product_id);
CREATE INDEX idx_orderitems_order_id ON Order_Items(order_id);

-- 5. For Q5 (Failed payments)
CREATE INDEX idx_payments_order_id ON Payments(order_id);
CREATE INDEX idx_payments_status ON Payments(payment_status);

-- 6. For Q6 (Top rated) and Q8 (Category performance)
CREATE INDEX idx_reviews_product_id ON Reviews(product_id);
CREATE INDEX idx_products_category_id ON Products(category_id);

-- 7. For Q7 (Delayed deliveries)
CREATE INDEX idx_deliveries_order_id ON Deliveries(order_id);
CREATE INDEX idx_deliveries_status ON Deliveries(delivery_status);

-- 8. Extra: For faster designer brand lookup
CREATE INDEX idx_designers_brand ON Designers(brand_name);
